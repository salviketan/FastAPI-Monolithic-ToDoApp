# syntax=docker/dockerfile:1

# https://docs.docker.com/engine/reference/builder/

ARG PYTHON_VERSION=3.14
# We declare the working directory as a variable to keep paths consistent
ARG APP_DIR=/todo_app/backend

# ==============================================================================
# Stage 1: Build & Dependency Installation
# ==============================================================================
FROM python:${PYTHON_VERSION}-slim-bookworm AS builder
ARG APP_DIR

# Install uv instantly using the official installer image
COPY --from=ghcr.io/astral-sh/uv:latest /uv /uvx /bin/

WORKDIR ${APP_DIR}
ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

# Install system dependencies needed for compiling packages (postgresql-dev, gcc)
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    libpq-dev \
    && rm -rf /var/lib/apt/lists/*

# Copy your native uv configuration files
COPY pyproject.toml uv.lock .python-version ${APP_DIR}/

# Install production dependencies only into a virtual environment (.venv)
RUN uv sync --no-dev --frozen

# ==============================================================================
# Stage 2: Final Production Runtime Image
# ==============================================================================
FROM python:${PYTHON_VERSION}-slim-bookworm AS runner
ARG APP_DIR
ARG UID=10001

WORKDIR ${APP_DIR}
ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

# Automatically inject the virtual environment into the system PATH
ENV PATH="${APP_DIR}/.venv/bin:$PATH"

# Install only the runtime dependency for postgresql (not the dev headers)
RUN apt-get update && apt-get install -y --no-install-recommends \
    libpq5 \
    && rm -rf /var/lib/apt/lists/*

# CRITICAL STEP: Copy ONLY the pre-compiled .venv from the builder stage
COPY --from=builder ${APP_DIR}/.venv ${APP_DIR}/.venv

# Copy your application code (Ensure you have a .dockerignore to skip local .venv)
COPY . ${APP_DIR}
